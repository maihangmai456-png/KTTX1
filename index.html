📋 Mã Google Apps Script Tích Hợp Google Sheets

Mã này dùng để dán vào Tiện ích mở rộng > Apps Script trong Google Sheets. Mã tự động tạo 2 sheet: DANH SACH HS (lưu thí sinh đăng nhập) và KET QUA KIEM TRA TX (lưu 14 cột điểm và kết quả bài làm). 

/**
 * =========================================================================
 * EDUSECURE - HỆ THỐNG THI TRẮC NGHIỆM CÔNG NGHỆ 10
 * MÃ GOOGLE APPS SCRIPT KẾT NỐI VỚI GOOGLE SHEETS
 * Tác giả: GV Mai Hằng
 * =========================================================================
 */

function doPost(e) {
  try {
    var rawData = e.postData.contents;
    var data = JSON.parse(rawData);
    var sheetId = data.sheetId || "1DTIBxYMavsb2tF56hJLPeT4n3n7YTBKB6ckpjDzSrOs";
    var ss = SpreadsheetApp.openById(sheetId);

    // 1. GHI DANH SÁCH HỌC SINH ĐĂNG NHẬP
    if (data.action === "LOGIN") {
      var logSheet = ss.getSheetByName("DANH SACH HS");
      if (!logSheet) {
        logSheet = ss.insertSheet("DANH SACH HS");
        logSheet.appendRow(["Thời Gian Đăng Nhập", "Mã / STT", "Họ Và Tên", "Lớp Học", "Loại Đề Thi"]);
        logSheet.getRange(1, 1, 1, 5).setFontWeight("bold").setBackground("#e0f2fe");
      }
      logSheet.appendRow([
        data.loginTime || new Date().toLocaleString("vi-VN"),
        data.studentCode || "",
        data.fullName || "",
        data.className || "",
        data.testType || ""
      ]);
      return ContentService.createTextOutput(JSON.stringify({ status: "success", action: "LOGIN" }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // 2. GHI KẾT QUẢ BÀI THI (14 CỘT CHUẨN)
    if (data.action === "SUBMISSION") {
      var resSheet = ss.getSheetByName("KET QUA KIEM TRA TX");
      if (!resSheet) {
        resSheet = ss.insertSheet("KET QUA KIEM TRA TX");
        resSheet.appendRow([
          "Thời Gian Nộp", "STT Thí Sinh", "Họ Và Tên Thí Sinh", "Lớp", "Tên Đề Thi", "Mã Đề",
          "Lần Thi", "Số Lần Vi Phạm", "Thời Gian Làm Bài", "Điểm Số", "Đúng Nhận Biết",
          "Đúng Thông Hiểu", "Đúng Vận Dụng", "Kết QuẢ Đúng/Sai (4 ý)"
        ]);
        resSheet.getRange(1, 1, 1, 14).setFontWeight("bold").setBackground("#dcfce7");
      }
      resSheet.appendRow([
        data.finishTime || new Date().toLocaleString("vi-VN"),
        data.studentCode || "",
        data.fullName || "",
        data.className || "",
        data.testName || "",
        data.examCode || "",
        data.attemptLabel || "Lần 1: Chính thức",
        data.violationsCount !== undefined ? data.violationsCount : 0,
        data.timeSpentFormatted || "",
        data.score !== undefined ? data.score : 0,
        data.correctNhanBiet !== undefined ? data.correctNhanBiet : 0,
        data.correctThongHieu !== undefined ? data.correctThongHieu : 0,
        data.correctVanDung !== undefined ? data.correctVanDung : 0,
        data.tfResultText || ""
      ]);
      return ContentService.createTextOutput(JSON.stringify({ status: "success", action: "SUBMISSION" }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Hành động không xác định" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
